// ---------- src/main.jsx ---------- import React from "react"; import ReactDOM from "react-dom/client"; import App from "./App.jsx"; import "./index.css";

ReactDOM.createRoot(document.getElementById("root")).render( <React.StrictMode> <App /> </React.StrictMode> );

// ---------- src/index.css ---------- @tailwind base; @tailwind components; @tailwind utilities;

body { background: #0f172a; color: white; }

// ---------- src/App.jsx ---------- import WalletConnect from "./components/WalletConnect"; import Hero from "./components/Hero"; import About from "./components/About"; import TokenInfo from "./components/TokenInfo"; import Roadmap from "./components/Roadmap"; import Footer from "./components/Footer";

function App() { return ( <div> <WalletConnect /> <Hero /> <About /> <TokenInfo /> <Roadmap /> <Footer /> </div> ); }

export default App;

// ---------- src/components/WalletConnect.jsx ---------- import { useState } from "react"; import { ethers } from "ethers";

export default function WalletConnect() { const [account, setAccount] = useState("");

async function connectWallet() { if (!window.ethereum) { alert("Install MetaMask"); return; }

const provider = new ethers.BrowserProvider(window.ethereum);
const accounts = await provider.send("eth_requestAccounts", []);
setAccount(accounts[0]);

}

return ( <div className="p-4 flex justify-between"> <h1 className="text-xl font-bold">Orakzai Bond</h1> <button onClick={connectWallet} className="bg-blue-500 px-4 py-2 rounded"> {account ? account.slice(0,6) + "..." : "Connect Wallet"} </button> </div> ); }

// ---------- src/components/Hero.jsx ---------- export default function Hero() { return ( <div className="text-center py-20"> <h2 className="text-4xl font-bold">Orakzai Bond</h2> <p className="mt-4">Next Generation Crypto Investment Ecosystem</p> </div> ); }

// ---------- src/components/About.jsx ---------- export default function About() { return ( <div className="p-10"> <h2 className="text-2xl font-bold">About</h2> <p className="mt-4"> Orakzai Bond is a decentralized financial platform designed for staking, investment pools and token utilities. </p> <h3 className="mt-6 font-bold">Mission</h3> <p>To empower global investors with blockchain-based opportunities.</p> <h3 className="mt-4 font-bold">Vision</h3> <p>To become a leading Web3 financial ecosystem.</p> </div> ); }

// ---------- src/components/TokenInfo.jsx ---------- import { useEffect, useState } from "react"; import { ethers } from "ethers";

const contractAddress = "0x6F539e4232c045cCAc08e2009d97BdC72815472a";

const abi = [ "function name() view returns (string)", "function symbol() view returns (string)", "function totalSupply() view returns (uint256)" ];

export default function TokenInfo() { const [data, setData] = useState({});

useEffect(() => { async function fetchData() { if (!window.ethereum) return; const provider = new ethers.BrowserProvider(window.ethereum); const contract = new ethers.Contract(contractAddress, abi, provider);

const name = await contract.name();
  const symbol = await contract.symbol();
  const totalSupply = await contract.totalSupply();

  setData({ name, symbol, totalSupply: totalSupply.toString() });
}

fetchData();

}, []);

return ( <div className="p-10"> <h2 className="text-2xl font-bold">Token Details</h2> <p>Name: {data.name}</p> <p>Symbol: {data.symbol}</p> <p>Total Supply: {data.totalSupply}</p> <p>Contract: {contractAddress}</p> </div> ); }

// ---------- src/components/Roadmap.jsx ---------- export default function Roadmap() { return ( <div className="p-10"> <h2 className="text-2xl font-bold">Roadmap</h2> <ul className="mt-4 list-disc ml-6"> <li>Phase 1: Launch & Community</li> <li>Phase 2: Staking & Lottery</li> <li>Phase 3: Exchange Listings</li> </ul> </div> ); }

// ---------- src/components/Footer.jsx ---------- export default function Footer() { return ( <div className="text-center p-6 border-t"> <p>© 2026 Orakzai Bond. All rights reserved.</p> </div> ); }
