<script>
  function calculateProfit() {
    const investors = document.getElementById("num_investors").value;
    const l1_reward = 10 * 0.05; // 5% of $10
    const l2_reward = 10 * 0.03; // 3% of $10
    const l3_reward = 10 * 0.02; // 2% of $10

    // Simplified calculation (assuming each person invites 1 more person)
    const l1_profit = investors * l1_reward;
    const l2_profit = investors * l2_reward;
    const l3_profit = investors * l3_reward;
    const total_profit = l1_profit + l2_profit + l3_profit;

    document.getElementById("l1_profit").innerText = "$" + l1_profit.toFixed(2);
    document.getElementById("l2_profit").innerText = "$" + l2_profit.toFixed(2);
    document.getElementById("l3_profit").innerText = "$" + l3_profit.toFixed(2);
    document.getElementById("total_profit").innerText = "$" + total_profit.toFixed(2);
  }
</script>
